apim/policies/payer-inbound.xml   -   $export DENY  +  MANDATORY CONTRACT STAMPING

<!-- $export is explicitly denied here so that a compromised ingest
     credential cannot be turned into a bulk extraction credential. -->
<choose>
  <when condition="@(context.Request.Url.Path.ToLower().Contains("/$export")
                  || context.Request.Url.Path.ToLower().Contains("/_operations/export"))">
    <return-response>
      <set-status code="403" reason="Forbidden" />
      ... "diagnostics": "$export is not available on the inbound ingest
          endpoint. Inbound and outbound are separately authorised routes
          over the same FHIR service."
    </return-response>
  </when>
</choose>

<!-- Every resource written through this endpoint is tagged with the contract
     it arrived under. This is what makes the outbound _tag filter meaningful:
     if ingest does not stamp, export cannot filter. -->
<choose>
  <when condition="@(context.Request.Method == "POST" || context.Request.Method == "PUT")">

    <!-- Refusing untagged writes is deliberate: silently untagged data is
         invisible to every payer and effectively lost. -->
    <choose>
      <when condition="@(!context.Request.Headers.ContainsKey("X-Payer-Contract"))">
        <return-response><set-status code="400" reason="Bad Request" /> ... </return-response>
      </when>
    </choose>

    <set-body>@{
        var body     = context.Request.Body.As<JObject>(preserveContent: true);
        var contract = context.Request.Headers.GetValueOrDefault("X-Payer-Contract","");
        var meta     = body["meta"] as JObject ?? new JObject();
        var tags     = meta["tag"]  as JArray  ?? new JArray();
        tags.Add(new JObject(
            new JProperty("system", "https://northwind.org/fhir/contract"),
            new JProperty("code",   contract)));
        ...
    }</set-body>
  </when>
</choose>

<rate-limit-by-key calls="3000" renewal-period="60" ... />

   Ingest stamps the contract; export filters on it. The two halves are one
   control. Assertions 13b and 14 depend on this stamping being mandatory.
